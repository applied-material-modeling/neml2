################################################################################
## Metallic fuel slug -- NEML2 on the OLD (legacy) mechanics system (prototype).
## Same as the original x447_dp11_fuel_moose.i mechanics (strain=FINITE, old
## tensor-mechanics), but the stress is supplied by our NEML2 model via the new
## ComputeNEML2StressOldSystem bridge material. Goal: reproduce the ORIGINAL
## old-system deck's numbers with NEML2 (native creep == NEML2 creep).
################################################################################

fuel_radius   = 2.1971e-03
fuel_area     = ${fparse pi * fuel_radius^2}
energy_in_fuel = 0.95
clad_to_fuel_htc = 6.104e4
run_time      = 56465640
max_time_step = 7200

[GlobalParams]
  displacements = 'disp_x disp_y'
[]

[Mesh]
  [fmg]
    type = FileMeshGenerator
    file = fuel_mesh.e
  []
  coord_type = RZ
[]

[Variables]
  [temp]
    initial_condition = 298
  []
  [disp_x]
  []
  [disp_y]
  []
[]

[AuxVariables]
  [burnup]
    order = FIRST
    family = LAGRANGE
  []
[]

[Functions]
  [power_history]
    type = PiecewiseLinear
    data_file = 'data/power_history.csv'
    format = rows
  []
  [peaking_power]
    type = PiecewiseLinear
    data_file = 'data/peaking_power_axial.csv'
    format = rows
    axis = y
  []
  [heat_source_func]
    type = ParsedFunction
    symbol_names  = 'p pk'
    symbol_values = 'power_history peaking_power'
    expression = '${fparse energy_in_fuel / fuel_area} * p * pk'
  []
  [clad_od_temp]
    type = PiecewiseMultilinear
    data_file = 'data/clad_od_temp_grid.txt'
  []
  [burnup_fima]
    type = PiecewiseLinear
    data_file = 'data/burnup_history.csv'
    format = rows
  []
  [burnup_field]
    type = ParsedFunction
    symbol_names  = 'b pk'
    symbol_values = 'burnup_fima peaking_power'
    expression = '100.0 * b * pk'
  []
[]

[Physics/SolidMechanics/QuasiStatic]
  [fuel]
    block = fuel
    strain = FINITE
    add_variables = false
    use_automatic_differentiation = false
    volumetric_locking_correction = true
    eigenstrain_names = 'thermal_eigenstrain swelling_eigenstrain'
    generate_output = 'stress_xx stress_yy stress_zz vonmises_stress'
  []
[]

[NEML2]
  eager = true
  input = 'fuel_creep_neml2.i.txt'
  [all]
    model = 'model'
    device = 'cpu'
    derivatives = 'neml2_stress neml2_strain'
  []
[]

[Kernels]
  [heat]
    type = ADHeatConduction
    variable = temp
  []
  [heat_ie]
    type = ADHeatConductionTimeDerivative
    variable = temp
  []
  [heat_source]
    type = HeatSource
    variable = temp
    function = heat_source_func
  []
[]

[AuxKernels]
  [burnup_aux]
    type = FunctionAux
    variable = burnup
    function = burnup_field
    execute_on = 'initial timestep_begin timestep_end'
  []
[]

[BCs]
  [no_x]
    type = DirichletBC
    variable = disp_x
    boundary = 'centerline'
    value = 0.0
    preset = true
  []
  [no_y]
    type = DirichletBC
    variable = disp_y
    boundary = 'fuel_bottom'
    value = 0.0
    preset = true
  []
  [surf_conv]
    type = ConvectiveFluxFunction
    variable = temp
    boundary = 'fuel_outer_radial_surface fuel_top fuel_bottom'
    T_infinity = clad_od_temp
    coefficient = ${clad_to_fuel_htc}
  []
[]

[Materials]
  [thermal_conductivity]
    type = ADParsedMaterial
    property_name = thermal_conductivity
    coupled_variables = 'temp'
    expression = '16.0 + 0.0151 * temp'
  []
  [thermal_other]
    type = ADGenericConstantMaterial
    prop_names  = 'specific_heat density'
    prop_values = '200.0      15800.0'
  []
  [thermal_expansion]
    type = ComputeThermalExpansionEigenstrain
    temperature = temp
    stress_free_temperature = 298.0
    thermal_expansion_coeff = 1.7e-5
    eigenstrain_name = thermal_eigenstrain
  []
  [swelling_volumetric]
    type = ParsedMaterial
    property_name = swelling_volumetric
    coupled_variables = 'burnup'
    expression = 'if(burnup <= 1.5, 0.24*burnup, 0.36 + 0.015*(burnup - 1.5))'
  []
  [swelling_eigenstrain]
    type = ComputeVolumetricEigenstrain
    volumetric_materials = 'swelling_volumetric'
    eigenstrain_name = swelling_eigenstrain
    args = 'burnup'
  []
  # --- feed NEML2 the mechanical strain; take back its stress into the OLD system ---
  [convert_strain]
    type = RankTwoTensorToSymmetricRankTwoTensor
    from = 'mechanical_strain'
    to = 'neml2_strain'
  []
  [stress]
    type = ComputeNEML2StressOldSystem
    neml2_stress = 'neml2_stress'
    neml2_jacobian = 'dneml2_stress/dneml2_strain'
  []
[]

[Preconditioning]
  [smp]
    type = SMP
    full = true
  []
[]

[Executioner]
  type = Transient
  solve_type = 'NEWTON'
  residual_and_jacobian_together = true
  petsc_options_iname = '-pc_type -pc_factor_mat_solver_type'
  petsc_options_value = 'lu       superlu_dist'
  automatic_scaling = true
  compute_scaling_once = false
  line_search = 'none'
  l_max_its = 50
  nl_max_its = 20
  nl_rel_tol = 1e-4
  nl_abs_tol = 1e-6
  end_time = ${run_time}
  dtmin = 1
  dtmax = ${max_time_step}
  [TimeStepper]
    type = IterationAdaptiveDT
    dt = 1e2
    optimal_iterations = 10
    iteration_window = 4
    timestep_limiting_function = power_history
    force_step_every_function_point = true
    max_function_change = 300
  []
[]

[Postprocessors]
  [temp_fuel_avg]
    type = ElementAverageValue
    variable = temp
    block = 'fuel'
  []
  [temp_fuel_max]
    type = ElementExtremeValue
    variable = temp
    block = 'fuel'
  []
  [burnup_avg]
    type = ElementAverageValue
    variable = burnup
    block = 'fuel'
  []
  [vonmises_avg]
    type = ElementAverageValue
    variable = vonmises_stress
    block = 'fuel'
  []
[]

[Outputs]
  csv = true
  file_base = fuel_old_coupled_out
[]
